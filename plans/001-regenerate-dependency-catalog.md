## Why this matters

The manifest and lockfile pin Slang 1.3.10, but the bundled reuse catalog was generated with 1.3.9. Recognition deliberately rejects mismatched parser versions, so a reproducible installation disables dependency reuse and retains full-review charges. Regenerate the catalog from exactly the existing reviewed reference files; preserve the conservative mismatch fallback and the estimator's coefficients.

The audit's 626 passing unit tests used stale installed Slang 1.3.9. They are not evidence for the locked 1.3.10 dependency set. A fresh install should expose the existing catalog-version regression test before regeneration.

Run this comparison after generation. Its temporary baseline is trusted catalog data from the stamped repository revision, never a target-project module:

```sh
node --input-type=module <<'JS'
import assert from 'node:assert/strict';
import { spawnSync } from 'node:child_process';
import { mkdtempSync, writeFileSync, rmSync } from 'node:fs';
import { tmpdir } from 'node:os';
import { join, resolve } from 'node:path';
import { pathToFileURL } from 'node:url';
const path = 'src/application/dependency-reuse/catalog.ts';
const snapshot = spawnSync('git', ['show', `1e61ad0:${path}`], { encoding: 'utf8' });
assert.equal(snapshot.status, 0, snapshot.stderr);
const temporary = mkdtempSync(join(tmpdir(), 'audit-catalog-comparison-'));
try {
  const priorPath = join(temporary, 'catalog.ts');
  writeFileSync(priorPath, snapshot.stdout);
  const { BUNDLED_CATALOG: before } = await import(pathToFileURL(priorPath).href);
  const { BUNDLED_CATALOG: after } = await import(pathToFileURL(resolve(path)).href);
  const identities = catalog => catalog.files.map(file =>
    ['id', 'repository', 'tag', 'commit', 'path', 'sha256', 'license'].map(key => file[key]));
  const declarations = catalog => catalog.declarations.map(item =>
    [item.file, item.kind, item.name]);
  assert.equal(after.parserVersion, '1.3.10');
  assert.equal(after.files.length, 37);
  assert.equal(after.declarations.length, 37);
  assert.equal(after.normalizerVersion, before.normalizerVersion);
  assert.equal(after.version, before.version);
  assert.deepEqual(identities(after), identities(before));
  assert.deepEqual(declarations(after), declarations(before));
  assert.deepEqual(after.declarations, before.declarations,
    'Token counts or fingerprints changed: STOP for compatibility review.');
  console.log(JSON.stringify({
    parserVersion: after.parserVersion,
    digest: after.digest,
    grammarSetsChanged: JSON.stringify(after.grammarSets) !== JSON.stringify(before.grammarSets),
    referenceGrammarEvidenceChanged: JSON.stringify(after.files) !== JSON.stringify(before.files),
  }, null, 2));
} finally {
  rmSync(temporary, { recursive: true, force: true });
}
JS
```
